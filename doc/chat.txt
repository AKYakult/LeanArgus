import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;

/** 注册请求 */
public record RegisterRequest(
        @NotBlank(message = "用户名不能为空")
        @Size(max = 64, message = "用户名长度不能超过 64")
        String username,

        @Email(message = "邮箱格式非法")
        @NotBlank(message = "邮箱不能为空")
        @Size(max = 128, message = "邮箱长度不能超过 128")

5. 新建 common/security/UserContext.java
这是本次最关键的一个类，值得多讲一下 why：

Spring MVC 处理一个 HTTP 请求时，会从线程池里挑一个线程来处理它（不同请求可能用不同线程，同一线程处理完一个请求后会被复用去处理别的请求）。业务代码里经常需要知道"当前是谁在发请求"——比如 QA 模块要知道是谁在提问、文档模块要知道谁有权限看这份文件。

笨办法是：每个 Controller 方法都手动加一个参数，把当前用户从最外层一路传到最里层的 Service、Mapper——但这样写起来非常啰嗦，参数表会越来越长。

参考项目用的办法是 ThreadLocal：可以理解成"每个线程自带一个储物柜，柜子的钥匙全局唯一，但每个线程打开自己那把钥匙看到的东西是不一样的"。做法是：

请求一进来，JwtAuthenticationFilter（阶段⑤要写）解析出当前用户，往这个"储物柜"里放一份 AuthenticatedUser；
这次请求处理过程中，任何 Service、任何 Controller，不用传参，直接调 UserContext.get() 就能拿到当前用户；
请求处理完，finally 里调用 UserContext.clear() 清空储物柜——这一步绝对不能漏，否则线程被复用去处理下一个请求时，"储物柜"里还留着上一个用户的信息，会造成串号（把 A 用户的请求当成 B 用户处理），是很严重的安全漏洞。
java
package com.example.myargus.common.security;

/**
 * 请求级用户上下文，基于 ThreadLocal 实现。
 * 由 JwtAuthenticationFilter 在请求进入时设置，请求结束时清理。
 * 业务代码任意位置通过 get() 静态方法获取当前用户，无需传参。
 */
public final class UserContext {

    private static final ThreadLocal<AuthenticatedUser> CURRENT_USER = new ThreadLocal<>();

    private UserContext() {
    }

    /** 设置当前请求的用户信息（由 Filter 调用） */
    public static void set(AuthenticatedUser user) {
        CURRENT_USER.set(user);
    }

    /** 获取当前请求的用户信息，未认证时返回 null */
    public static AuthenticatedUser get() {
        return CURRENT_USER.get();
    }

    /** 清理当前线程的用户信息（由 Filter 在 finally 中调用，防止内存泄漏和数据污染） */
    public static void clear() {
        CURRENT_USER.remove();
    }
}

2. 新建 auth/config/AuthProperties.java
为什么 JWT 密钥、token 有效期这些不能写死在代码里:

密钥这种敏感信息一旦硬编码进 .java 文件,就会被提交到 Git 仓库里,任何能看到源码的人都能拿着这个密钥伪造登录凭证。正确做法是放进 application.yaml(或者更进一步,放到环境变量/密钥管理服务里,application.yaml 里只写占位符),然后用 Spring 的 @ConfigurationProperties 机制自动绑定成一个 Java 对象。

这样做还有个好处:开发环境、测试环境、生产环境可以用不同的配置文件(application-dev.yaml / application-prod.yaml),密钥、过期时间都能按环境单独设置,代码完全不用改。

java
package com.example.myargus.auth.config;

import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotBlank;
import lombok.Data;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.validation.annotation.Validated;

/**
 * 认证相关配置属性,前缀 {@code myargus.auth}。
 */
@Validated
@ConfigurationProperties(prefix = "myargus.auth")
@Data
public class AuthProperties {

    /** JWT 签发者 */
    @NotBlank
    private String issuer = "myargus";

    /** Access token 有效期（分钟） */
    @Min(1)
    private int accessTokenExpireMinutes = 30;

    /** Refresh token 有效期（天） */
    @Min(1)
    private int refreshTokenExpireDays = 14;

    /** JWT HMAC 签名密钥,至少 32 字节 */
    @NotBlank
    private String jwtSecret;

    /** Refresh token Cookie 名称 */
    @NotBlank
    private String refreshCookieName = "MYARGUS_REFRESH_TOKEN";

    /** Refresh token Cookie 是否仅 HTTPS 发送（生产环境应设为 true） */
    private boolean refreshCookieSecure = true;
}
说明一处有意的差异:参考项目配置前缀用的是 argus.rag.auth,我这里改成了 myargus.auth——配置前缀是自己起的命名空间,不需要跟包名/类名严格一致,用你自己项目的名字更清晰。如果你想跟参考项目完全一致也可以,把下面 yaml 和这里的 prefix 都改成 argus.rag.auth 即可,不影响任何逻辑。

3. 新建 auth/config/AuthConfiguration.java
为什么要在这里同时注册 Clock 这个 bean:

Clock.systemDefaultZone() 就是"现在几点"的意思,但如果代码里到处直接写 LocalDateTime.now(),写单元测试时就没法控制"现在"是哪个时间点(比如想测试"token 过期后能不能用"这种场景,总不能真等 14 天)。把 Clock 做成一个可注入的 bean,测试时可以传一个"固定在某个时间点"的假 Clock 进去,时间相关的逻辑就能被精确控制和测试了。这是个很实用的技巧,值得记住。

这个类同时把原来 SecurityConfig 里的密码加密 bean 也接管过来了:

java
package com.example.myargus.auth.config;

import com.example.myargus.common.exception.BusinessException;
import com.example.myargus.auth.service.PasswordHasher;
import org.springframework.boot.context.properties.EnableConfigurationProperties;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;

import java.nio.charset.StandardCharsets;
import java.time.Clock;

/**
 * 认证模块配置,注册 PasswordHasher 和 Clock Bean。
 */
@Configuration
@EnableConfigurationProperties(AuthProperties.class)
public class AuthConfiguration {

    /** BCrypt 最大输入字节数 */
    private static final int BCRYPT_MAX_INPUT_BYTES = 72;

    @Bean
    Clock authClock() {
        return Clock.systemDefaultZone();
    }

    /** 基于 BCrypt 的密码哈希实现 */
    @Bean
    PasswordHasher defaultPasswordHasher() {
        BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();
        return new PasswordHasher() {
            @Override
            public String hash(String rawPassword) {
                validateInputLength(rawPassword);
                return encoder.encode(rawPassword);
            }

            @Override
            public boolean matches(String rawPassword, String passwordHash) {
                validateInputLength(rawPassword);
                return encoder.matches(rawPassword, passwordHash);
            }

            private void validateInputLength(String rawValue) {
                if (rawValue == null || rawValue.getBytes(StandardCharsets.UTF_8).length > BCRYPT_MAX_INPUT_BYTES) {
                    throw new BusinessException("密码长度超过安全上限,请控制在 72 字节以内");
                }
            }
        };
    }
}
这里有个容易被忽略但很重要的细节:BCrypt 算法本身有个限制——只会处理密码的前 72 个字节,超出部分会被静默截断(不会报错,而是直接忽略)。如果不做这个校验,会出现一个隐蔽的 bug:用户设置一个 100 字节的密码,注册时存的哈希其实只对应前 72 字节;下次登录如果用户脑子里记的是完整密码,理论上后 28 字节不影响登录结果(因为校验时同样会截断到 72 字节),表面上没问题——但这种"隐性截断"仍然是不好的行为,提前用异常挡住比让用户稀里糊涂踩坑要好。

顺便解释一下为什么 refresh token 要专门存一张表,而不是也签成 JWT:

Access token(JWT)的特点是"自包含、无状态"——服务器不用查数据库就能验证它合不合法,靠的是签名。但这个特点也是它的缺陷:一旦签发出去,在过期之前没法主动吊销(除非你专门再搞一套黑名单机制,那还不如干脆用数据库)。

Refresh token 存活时间长(这里配的是 14 天),一旦泄露,风险比只活 30 分钟的 access token 大得多,所以必须支持"随时吊销"——比如用户点击"退出登录"、或者系统检测到异常要强制下线。存数据库、可以直接 UPDATE ... SET revoked_at = now() 就是最直接的解决方案。这也是为什么阶段①⑤都还没做,但数据库表你已经提前建好了。

package com.example.myargus.auth.mapper;

import com.example.myargus.auth.model.entity.UserRefreshToken;
import com.baomidou.mybatisplus.core.mapper.BaseMapper;
import org.apache.ibatis.annotations.Mapper;
import org.apache.ibatis.annotations.Param;
import org.apache.ibatis.annotations.Update;

import java.time.LocalDateTime;

@Mapper
public interface UserRefreshTokenMapper extends BaseMapper<UserRefreshToken> {

    /** 原子吊销:仅当 revoked_at 为 null 时才更新,返回受影响行数用于检测并发重放 */
    @Update("update user_refresh_tokens set revoked_at = #{revokedAt} where id = #{id} and revoked_at is null")
    int revokeByIdIfActive(@Param("id") Long id, @Param("revokedAt") LocalDateTime revokedAt);
}
这个 revokeByIdIfActive 方法为什么要写成"仅当 revoked_at 为 null 时才更新"而不是直接 update ... where id = ?:

这是在防一种叫 "refresh token 重放攻击" 的场景:假设一个 refresh token 被黑客窃取了。正常用户和黑客几乎同时各拿着同一个 token 去请求"刷新令牌"接口。如果不做原子控制,两个请求可能都读到"这个 token 还没吊销",都各自成功刷新一次,黑客就这样静默拿到了新的合法 token,而你完全不知道。

用 where id = ? and revoked_at is null 这种写法,数据库层面保证了"多个并发请求里只有第一个能成功把 revoked_at 从 null 改成有值",后面的请求执行这条 SQL 时,revoked_at is null 这个条件已经不成立了,返回的受影响行数是 0。阶段④写 AuthService.refresh() 时你会看到:一旦发现"应该能吊销却没吊销成功"(返回 0),就直接判定为疑似重放攻击,把这个用户的所有 token 全部吊销,强制重新登录。

MyBatis-Plus 的 BaseMapper 只提供通用的增删改查,像这种"带条件的原子更新"就得像这样自己写一条 @Update 原生 SQL——这跟你之前 UserMapper 继承 BaseMapper 的写法是完全一样的套路,只是多了一个自定义方法。
